Commands

cd /opt/airflow/wwi_snowflake/extract
python extract_load.py --all                                  # every table, incremental
python extract_load.py --table Sales.Customers                # one table
python reconcile.py                                           # source vs Snowflake
cd /opt/airflow/wwi_snowflake/dbt
dbt debug --profiles-dir . --target prod
dbt run --select silver --profiles-dir . --target prod
dbt snapshot --profiles-dir . --target prod
dbt run --select gold --profiles-dir . --target prod
dbt run --select gold --full-refresh --profiles-dir . --target prod    # rebuild incremental facts from scratch
dbt run --select path:models/gold/dimensions --profiles-dir . --target prod   # just one kind, by hand
dbt run --select mart_sales --profiles-dir . --target prod
dbt test --select gold --indirect-selection cautious --profiles-dir . --target prod
airflow dags trigger wwi_snowflake_daily
airflow dags list-runs -d wwi_snowflake_daily
dbt build --profiles-dir . --target prod                      # everything, in dependency order
select * from WWI_RAW.BRONZE._LOAD_AUDIT order by loaded_at desc;          -- what each load did
select * from WWI_RAW.BRONZE._LOAD_WATERMARK order by table_name;          -- where each table resumes
list @WWI_RAW.BRONZE.LANDING;                                              -- staged files (role LOADER)
select count_if(customer_key = -1), count(*) from WWI_ANALYTICS.GOLD.FACT_SALE;   -- unknown members
-- as LOADER: forget the table's mark, and optionally empty its Bronze table
delete from WWI_RAW.BRONZE._LOAD_WATERMARK where table_name = 'SALES__CUSTOMERS';
truncate table WWI_RAW.BRONZE.SALES__CUSTOMERS;
-- then: python extract_load.py --table Sales.Customers
